cmake_minimum_required(VERSION 3.18)
project(dnf5_plugin_repo_snapshot LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
add_library(snapshot_core STATIC snapshot.cpp)
target_include_directories(snapshot_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
set_target_properties(snapshot_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
option(SNAPSHOT_BUILD_PLUGINS "Build external DNF5 and libdnf5 modules" OFF)
if(SNAPSHOT_BUILD_PLUGINS)
    include(GNUInstallDirs)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(LIBDNF5 REQUIRED IMPORTED_TARGET libdnf5)
    pkg_check_modules(LIBDNF5_CLI REQUIRED IMPORTED_TARGET libdnf5-cli)
    add_library(repo_snapshot_cmd_plugin MODULE cli.cpp)
    target_link_libraries(repo_snapshot_cmd_plugin PRIVATE snapshot_core PkgConfig::LIBDNF5 PkgConfig::LIBDNF5_CLI)
    set_target_properties(repo_snapshot_cmd_plugin PROPERTIES PREFIX "")
    install(TARGETS repo_snapshot_cmd_plugin LIBRARY DESTINATION "${CMAKE_INSTALL_LIBDIR}/dnf5/plugins")
    add_library(repo_snapshot_plugin MODULE hook.cpp)
    target_link_libraries(repo_snapshot_plugin PRIVATE snapshot_core PkgConfig::LIBDNF5)
    set_target_properties(repo_snapshot_plugin PROPERTIES PREFIX "" OUTPUT_NAME "repo_snapshot")
    install(TARGETS repo_snapshot_plugin LIBRARY DESTINATION "${CMAKE_INSTALL_LIBDIR}/libdnf5/plugins")
    install(FILES repo_snapshot.conf DESTINATION "${CMAKE_INSTALL_SYSCONFDIR}/dnf/libdnf5-plugins")
endif()
if(BUILD_TESTING)
    add_executable(snapshot_tests tests.cpp)
    target_link_libraries(snapshot_tests PRIVATE snapshot_core)
    add_test(NAME snapshot_tests COMMAND snapshot_tests)
endif()
